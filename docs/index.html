public final class APIClient: @unchecked Sendable {
    public static let shared = APIClient()
    public var auth: APIAuth?

    public func send<T: Decodable>(
        path: String,
        method: HTTPMethod,
        queryItems: [URLQueryItem] = [],
        headers: [String: String] = [:],
        responseType: T.Type
    ) async throws -> T {
        let data = try await execute(
            path: path,
            method: method,
            queryItems: queryItems,
            headers: headers,
            bodyData: nil
        )
        return try decode(data, as: responseType)
    }
}
public func listTasks(
    workspaceID: String,
    status: TaskStatus? = nil,
    assigneeID: String? = nil,
    limit: Int? = nil,
    cursor: String? = nil
) async throws -> TaskPage {
    let path = "/workspaces/\(workspaceID)/tasks"
    var queryItems: [URLQueryItem] = []

    if let status {
        queryItems.append(.init(name: "status", value: status.rawValue))
    }
    if let limit {
        queryItems.append(.init(name: "limit", value: String(limit)))
    }

    return try await send(
        path: path,
        method: .get,
        queryItems: queryItems,
        responseType: TaskPage.self
    )
}

FAQ

Good to know before you build.

Is SpecForge really a native Mac app?

Yes. It’s built with SwiftUI and runs entirely on your Mac. There is no web view, local server, or account layer in the app.

Where do my credentials go?

They are attached only to the outgoing request that downloads the specification. Tokens and passwords are never embedded in generated files, logged, or sent anywhere else.

What Swift version does the generated code need?

The generated package uses Swift 5.9+ and targets iOS 15, macOS 12, tvOS 15, and watchOS 8 or later. The macOS app itself requires macOS 14+.

Can I use the output in an existing Xcode project?

Yes. Export a package and add it as a local dependency, or turn off “Swift Package” during generation to get flat Swift files you can drag into a target.

Does it handle private specification URLs?

Yes. Choose Bearer, API key, or Basic authentication before fetching. The URL flow supports protected endpoints and localhost.

Is it free?

The app is free to download. The source is available on GitHub so you can inspect the generator and build it yourself.

How do I install it?

Open the DMG, drag SpecForge to Applications, and launch it. The build is ad-hoc signed but not notarized, so on first launch macOS may ask you to confirm: right-click the app, choose Open, then confirm. You can also allow it later in System Settings → Privacy & Security.
